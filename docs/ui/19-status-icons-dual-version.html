4. 实现方案

版本检测 + 条件渲染
检测策略：运行时探测宿主状态点的 CSS 结构，而非硬编码版本号。

方法：在首次渲染时，创建一个临时测试元素，读取其 ::before 伪元素的 opacity 值。 若 opacity < 0.5，说明宿主使用双层结构（0.1.x）；否则为单层结构（0.2.0+）。

优点：
• 不依赖版本号（避免硬编码）
• 自动适配未来可能的再次变更
• 检测失败时降级到单层（更安全） 
// 检测宿主状态点结构
function detectDotStructure(): 'legacy' | 'modern' {
  const test = document.createElement('span');
  test.className = '_dot_1i3xo_2'; // 宿主状态点类名
  test.style.cssText = 'position:absolute;visibility:hidden';
  document.body.appendChild(test);
  
  const before = getComputedStyle(test, '::before');
  const hasGlow = before && parseFloat(before.opacity) < 0.5;
  document.body.removeChild(test);
  
  return hasGlow ? 'legacy' : 'modern';
}

// 条件渲染
const dotStructure = detectDotStructure();
const dotClass = dotStructure === 'legacy' 
  ? 'ss-hover-status-dot-legacy' 
  : 'ss-hover-status-dot';

5. CSS 差异对照

DSH 0.1.x 样式（legacy）
.ss-hover-status-dot-legacy {
  position: relative;
  display: block;
  flex: none;
  width: 10px;
  height: 10px;
  color: #adb2b8;
}
.ss-hover-status-dot-legacy::before,
.ss-hover-status-dot-legacy::after {
  content: "";
  position: absolute;
  border-radius: 50%;
  background: currentColor;
}
.ss-hover-status-dot-legacy::before {
  inset: 0;
  opacity: 0.1; /* 光晕 */
}
.ss-hover-status-dot-legacy::after {
  inset: 2px; /* 6px 核心 */
}
.ss-hover-status-dot-legacy[data-state="urgent"] {
  color: #f59e0b;
}
.ss-hover-status-dot-legacy[data-state="overdue"] {
  color: #f25a5a;
}
DSH 0.2.0+ 样式（modern）
.ss-hover-status-dot {
  position: relative;
  display: block;
  flex: none;
  width: 10px;
  height: 10px;
  color: #adb2b8;
}
.ss-hover-status-dot::after {
  content: "";
  position: absolute;
  inset: 20%; /* 6px 核心 */
  border-radius: 50%;
  background: currentColor;
  corner-shape: round; /* 新属性 */
}
/* 无 ::before 光晕层 */
.ss-hover-status-dot[data-state="urgent"] {

}
